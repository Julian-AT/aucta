    Define `sessionSummarize` as:
    ```typescript
    export const sessionSummarize = inngest.createFunction(
      { id: "session-summarize", retries: 3 },
      { event: "session/ended" },
      async ({ event, step }) => {
        const sessionData = await step.run("fetch-session", async () => {
          // Instantiate repos with db, fetch session + findings + tool calls
          const sessionRepo = new SessionRepository(db);
          return sessionRepo.findWithDetails(event.data.sessionId);
        });

        if (!sessionData) return { status: "not-found" };

        const summary = await step.run("generate-summary", async () => {
          return generateSessionSummary(sessionData);
        });

        await step.run("save-summary", async () => {
          const sessionRepo = new SessionRepository(db);
          return sessionRepo.updateSummary(event.data.sessionId, summary);
        });

        return { status: "completed", sessionId: event.data.sessionId };
      }
    );
    ```
