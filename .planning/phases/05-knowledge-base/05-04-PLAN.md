Create the ingest directory. The session ingester uses Gemini to extract structured learnings from session data.

```typescript
import { GoogleGenAI } from "@google/genai";

// Plain interface (not DTO) for Inngest JSON serialization compatibility
// (per Phase 04 decision: generateSessionSummary uses plain interface)
export interface SessionIngestionInput {
  session: {
    id: string;
    orgId: string;
    languageDetected: string | null;
    summary: string | null;
  };
  findings: Array<{
    category: string;
    description: string;
    severity: string;
    equipmentModel: string | null;
    manufacturer: string | null;
  }>;
  toolCalls: Array<{
    toolName: string;
    success: boolean;
  }>;
}

export interface CandidateArticle {
  title: string;
  contentMd: string;
  category: "equipment" | "procedure" | "safety" | "regulatory" | "troubleshooting";
  equipmentBrand: string | null;
  equipmentModel: string | null;
  procedureType: string | null;
  tags: string[];
}

export async function extractLearnings(
  data: SessionIngestionInput,
  apiKey: string,
): Promise<CandidateArticle[]> {
  // Skip sessions with no meaningful content
  if (data.findings.length === 0 && !data.session.summary) {
    return [];
  }

  const model = process.env.GEMINI_SUMMARY_MODEL || "gemini-2.5-flash";
  const ai = new GoogleGenAI({ apiKey });

  const findingsList = data.findings
    .map((f) => `- [${f.severity}] ${f.category}: ${f.description}${f.equipmentModel ? ` (${f.manufacturer ?? ""} ${f.equipmentModel})` : ""}`)
    .join("\n");

  const prompt = `You are extracting reusable knowledge from an HVAC technician's AI copilot session.
Analyze the session data below and output JSON array of knowledge article candidates.

SESSION SUMMARY:
${data.session.summary ?? "No summary available."}

FINDINGS LOGGED:
${findingsList || "No findings logged."}

TOOL CALLS: ${data.toolCalls.map((tc) => tc.toolName).join(", ") || "None"}

For each distinct piece of reusable knowledge, output a JSON object with:
- title: Clear, descriptive title (e.g., "Daikin FTXM25 E3 Error Code Resolution")
- contentMd: Detailed markdown content with steps, observations, and context. Include safety warnings where relevant.
- category: One of "equipment", "procedure", "safety", "regulatory", "troubleshooting"
- equipmentBrand: Brand name if identifiable (e.g., "Daikin"), null otherwise
- equipmentModel: Model number if identifiable (e.g., "FTXM25"), null otherwise
- procedureType: Type of procedure if applicable (e.g., "error_diagnosis", "compressor_replacement"), null otherwise
- tags: Array of relevant tags (e.g., ["error-code", "E3", "outdoor-unit"])

Rules:
- Only extract genuinely reusable knowledge, not trivial observations
- If the session has no meaningful learnings, return an empty array []
- Each article should be self-contained and useful to another technician facing the same issue
- Include safety disclaimers where relevant (refrigerant handling, electrical work)

Respond with ONLY the JSON array, no surrounding text.`;

  const result = await ai.models.generateContent({ model, contents: prompt });
  const text = result.text ?? "[]";

  try {
    const parsed = JSON.parse(text);
    if (!Array.isArray(parsed)) return [];
    return parsed.filter(
      (item: unknown): item is CandidateArticle =>
        typeof item === "object" &&
        item !== null &&
        "title" in item &&
        "contentMd" in item &&
        "category" in item,
    );
  } catch {
    return [];
  }
}
```

**File 2: packages/api/src/jobs/knowledge-ingest.ts**

Follow the session-summarize.ts Inngest pattern exactly. Use concurrency control per D-01 and Pitfall 6.

```typescript
import { KnowledgeArticleRepository } from "@aucta/db/repositories/knowledge-article.repository";
import { SessionRepository } from "@aucta/db/repositories/session.repository";
import { GeminiEmbeddingService } from "@aucta/ai/knowledge";
import { createQdrantClient } from "@aucta/ai/knowledge";
import { QDRANT_CONFIG, EMBEDDING_CONFIG } from "@aucta/ai/knowledge";
import { db } from "@aucta/db/client";
import { extractLearnings } from "@aucta/ai/knowledge/ingest/session-ingester";
import { inngest } from "./inngest-client.js";

export const knowledgeIngest = inngest.createFunction(
  {
    id: "knowledge-ingest",
    retries: 3,
    concurrency: [{ key: "event.data.orgId", limit: 1 }],
    triggers: [{ event: "session/ended" }],
  },
  async ({ event, step }) => {
    // Step 1: Fetch session with details
    const sessionData = await step.run("fetch-session", async () => {
      const sessionRepo = new SessionRepository(db);
      return sessionRepo.findWithDetails(event.data.sessionId);
    });

    if (!sessionData) return { status: "not-found" };

    // Step 2: Extract candidate learnings via LLM
    const candidates = await step.run("extract-learnings", async () => {
      const apiKey = process.env.GEMINI_API_KEY;
      if (!apiKey) throw new Error("GEMINI_API_KEY not set");

      return extractLearnings(
        {
          session: {
            id: sessionData.session.id,
            orgId: sessionData.session.orgId,
            languageDetected: sessionData.session.languageDetected,
            summary: sessionData.session.summary,
          },
          findings: sessionData.findings.map((f) => ({
            category: f.category,
            description: f.description,
            severity: f.severity,
            equipmentModel: f.equipmentModel,
            manufacturer: f.manufacturer,
          })),
          toolCalls: sessionData.toolCalls.map((tc) => ({
            toolName: tc.toolName,
            success: tc.success,
          })),
        },
        apiKey,
      );
    });

    if (candidates.length === 0) return { status: "no-learnings" };

    // Step 3: Create articles in Postgres and embed in Qdrant
    const articleIds = await step.run("create-articles", async () => {
      const articleRepo = new KnowledgeArticleRepository(db);
      const apiKey = process.env.GEMINI_API_KEY!;
      const qdrantUrl = process.env.QDRANT_URL;
      const qdrantApiKey = process.env.QDRANT_API_KEY;

      const ids: string[] = [];
      for (const candidate of candidates) {
        const article = await articleRepo.create({
          orgId: sessionData.session.orgId,
          title: candidate.title,
          contentMd: candidate.contentMd,
          category: candidate.category,
          equipmentBrand: candidate.equipmentBrand ?? undefined,
          equipmentModel: candidate.equipmentModel ?? undefined,
          procedureType: candidate.procedureType ?? undefined,
          sourceType: "session",
          sourceSessionIds: [event.data.sessionId],
          tags: candidate.tags,
          isAiGenerated: true,
        });

        // Embed and upsert to Qdrant if configured
        if (qdrantUrl) {
          try {
            const embeddingService = new GeminiEmbeddingService({ apiKey });
            const qdrantClient = createQdrantClient({ url: qdrantUrl, apiKey: qdrantApiKey });
            const embeddingText = `${article.title} ${article.summary ?? ""} ${article.contentMd}`;
            const denseVector = await embeddingService.embedDocument(embeddingText);

            await qdrantClient.upsert(QDRANT_CONFIG.COLLECTION_NAME, {
              wait: true,
              points: [
                {
                  id: article.id,
                  vectors: {
                    [QDRANT_CONFIG.DENSE_VECTOR_NAME]: denseVector,
                    [QDRANT_CONFIG.SPARSE_VECTOR_NAME]: {
                      text: embeddingText,
                      model: QDRANT_CONFIG.BM25_MODEL,
                    },
                  },
                  payload: {
                    article_id: article.id,
                    org_id: article.orgId,
                    confidence_level: article.confidenceLevel,
                    equipment_brand: article.equipmentBrand ?? "",
                    equipment_model: article.equipmentModel ?? "",
                    category: article.category,
                    title: article.title,
                    summary: article.summary ?? "",
                    tags: article.tags ?? [],
                  },
                },
              ],
            });

            await articleRepo.updateEmbedding(article.id, article.id, EMBEDDING_CONFIG.MODEL);
          } catch (err) {
            // Log but don't fail -- article is in Postgres, embedding can be retried
            console.error(`Failed to embed article ${article.id}:`, err);
          }
        }

        ids.push(article.id);
      }
      return ids;
    });

    // Step 4: Send compilation event for merging related articles
    await step.run("trigger-compilation", async () => {
      await inngest.send({
        name: "knowledge/articles-ingested",
        data: {
          orgId: sessionData.session.orgId,
          articleIds,
          sessionId: event.data.sessionId,
        },
      });
    });

    return { status: "completed", articleCount: candidates.length, articleIds };
  },
);
```

Note: The exact Qdrant upsert API for sparse vectors with server-side inference may need adapting based on the @qdrant/js-client-rest types. The `text` + `model` fields in the sparse vector may need to be wrapped differently. Read the SDK types to confirm.

```typescript
import { GoogleGenAI } from "@google/genai";

export interface ArticleForCompilation {
  id: string;
  title: string;
  contentMd: string;
  category: string;
  equipmentBrand: string | null;
  equipmentModel: string | null;
  procedureType: string | null;
  sourceSessionIds: string[] | null;
  version: number;
}

export interface CompiledArticle {
  title: string;
  summary: string;
  contentMd: string;
  tags: string[];
  sourceSessionIds: string[];
}

/**
 * Merge multiple articles about the same equipment/procedure into one authoritative article.
 * Uses LLM to synthesize observations into a comprehensive wiki-style article.
 */
export async function compileArticle(
  articles: ArticleForCompilation[],
  apiKey: string,
): Promise<CompiledArticle> {
  if (articles.length === 0) {
    throw new Error("Cannot compile zero articles");
  }

  // Single article -- just generate a summary
  if (articles.length === 1) {
    const article = articles[0]!;
    return {
      title: article.title,
      summary: await generateSummary(article.contentMd, apiKey),
      contentMd: article.contentMd,
      tags: [],
      sourceSessionIds: article.sourceSessionIds ?? [],
    };
  }

  const model = process.env.GEMINI_SUMMARY_MODEL || "gemini-2.5-flash";
  const ai = new GoogleGenAI({ apiKey });

  const articlesText = articles
    .map((a, i) => `--- Article ${i + 1}: "${a.title}" ---\n${a.contentMd}`)
    .join("\n\n");

  const equipment = [articles[0]!.equipmentBrand, articles[0]!.equipmentModel]
    .filter(Boolean)
    .join(" ");

  const prompt = `You are compiling multiple knowledge articles about ${equipment || "an HVAC topic"} into one comprehensive, authoritative wiki article.

SOURCE ARTICLES:
${articlesText}

Combine all observations, procedures, and learnings into ONE structured article. Requirements:
1. Merge duplicate information, keeping the most detailed version
2. Organize with clear markdown headers (##, ###)
3. Include all safety warnings from source articles
4. Preserve specific model numbers, error codes, and measurements
5. Write in a professional, instructional tone suitable for HVAC technicians
6. Add a "Summary" (1-2 sentences) at the top

Respond with JSON:
{
  "title": "Clear descriptive title",
  "summary": "1-2 sentence summary",
  "contentMd": "Full markdown content",
  "tags": ["relevant", "tags"]
}`;

  const result = await ai.models.generateContent({ model, contents: prompt });
  const text = result.text ?? "{}";

  try {
    const parsed = JSON.parse(text);
    const allSessionIds = articles.flatMap((a) => a.sourceSessionIds ?? []);
    return {
      title: parsed.title ?? articles[0]!.title,
      summary: parsed.summary ?? "",
      contentMd: parsed.contentMd ?? articles[0]!.contentMd,
      tags: Array.isArray(parsed.tags) ? parsed.tags : [],
      sourceSessionIds: [...new Set(allSessionIds)],
    };
  } catch {
    // Fallback: return first article with combined session IDs
    const allSessionIds = articles.flatMap((a) => a.sourceSessionIds ?? []);
    return {
      title: articles[0]!.title,
      summary: "",
      contentMd: articles[0]!.contentMd,
      tags: [],
      sourceSessionIds: [...new Set(allSessionIds)],
    };
  }
}

async function generateSummary(contentMd: string, apiKey: string): Promise<string> {
  const model = process.env.GEMINI_SUMMARY_MODEL || "gemini-2.5-flash";
  const ai = new GoogleGenAI({ apiKey });

  const result = await ai.models.generateContent({
    model,
    contents: `Summarize this HVAC knowledge article in 1-2 sentences:\n\n${contentMd}`,
  });

  return result.text ?? "";
}
```

```typescript
import type { KnowledgeArticleRepository } from "@aucta/db/repositories/knowledge-article.repository";
import type { KnowledgeLinkRepository } from "@aucta/db/repositories/knowledge-link.repository";

export interface LinkerDeps {
  articleRepo: KnowledgeArticleRepository;
  linkRepo: KnowledgeLinkRepository;
}

/**
 * Auto-generate "related" links between articles that share the same equipment brand.
 * Clears existing links for the source article, then creates fresh links (per D-09).
 */
export async function generateLinks(
  deps: LinkerDeps,
  articleId: string,
  orgId: string,
): Promise<number> {
  const article = await deps.articleRepo.findById(articleId);
  if (!article || !article.equipmentBrand) return 0;

  // Find other articles for the same equipment brand in this org
  const relatedArticles = await deps.articleRepo.findByEquipment(
    orgId,
    article.equipmentBrand,
    undefined, // all models for this brand
  );

  // Remove existing outgoing links for this article (idempotent re-linking)
  await deps.linkRepo.deleteByArticleId(articleId);

  let linkCount = 0;
  for (const related of relatedArticles) {
    if (related.id === articleId) continue; // skip self

    await deps.linkRepo.create({
      sourceArticleId: articleId,
      targetArticleId: related.id,
      linkType: related.equipmentModel === article.equipmentModel ? "related" : "reference",
    });
    linkCount++;
  }

  return linkCount;
}
```

```typescript
import { db } from "@aucta/db/client";
import { inngest } from "./inngest-client.js";

export const knowledgeCompile = inngest.createFunction(
  {
    id: "knowledge-compile",
    retries: 3,
    concurrency: [{ key: "event.data.orgId", limit: 1 }],
    triggers: [{ event: "knowledge/articles-ingested" }],
  },
  async ({ event, step }) => {
    const { orgId, articleIds } = event.data as { orgId: string; articleIds: string[] };

    // Step 1: Check for existing articles that could be merged
    const mergeTargets = await step.run("find-merge-targets", async () => {
      const articleRepo = new KnowledgeArticleRepository(db);
      const results: Array<{ articleId: string; existingIds: string[] }> = [];

      for (const articleId of articleIds) {
        const article = await articleRepo.findById(articleId);
        if (!article || !article.equipmentBrand) continue;

        const existing = await articleRepo.findByEquipment(
          orgId,
          article.equipmentBrand,
          article.equipmentModel ?? undefined,
        );

        // If there are other articles for the same equipment, they're merge candidates
        const otherIds = existing
          .filter((e) => e.id !== articleId)
          .map((e) => e.id);

        if (otherIds.length > 0) {
          results.push({ articleId, existingIds: otherIds });
        }
      }

      return results;
    });

    // Step 2: Compile merged articles where applicable
    const compiledIds = await step.run("compile-articles", async () => {
      const apiKey = process.env.GEMINI_API_KEY;
      if (!apiKey) throw new Error("GEMINI_API_KEY not set");

      const articleRepo = new KnowledgeArticleRepository(db);
      const compiled: string[] = [];

      for (const target of mergeTargets) {
        const allIds = [target.articleId, ...target.existingIds];
        const articles = await Promise.all(
          allIds.map((id) => articleRepo.findById(id)),
        );

        const validArticles = articles.filter(
          (a): a is NonNullable<typeof a> => a !== null,
        );

        if (validArticles.length < 2) continue;

        const result = await compileArticle(
          validArticles.map((a) => ({
            id: a.id,
            title: a.title,
            contentMd: a.contentMd,
            category: a.category,
            equipmentBrand: a.equipmentBrand,
            equipmentModel: a.equipmentModel,
            procedureType: a.procedureType,
            sourceSessionIds: a.sourceSessionIds,
            version: a.version,
          })),
          apiKey,
        );

        // Update the newest article with compiled content
        const primaryArticle = validArticles[0]!;
        await articleRepo.update(primaryArticle.id, {
          title: result.title,
          summary: result.summary,
          contentMd: result.contentMd,
          tags: result.tags,
          version: primaryArticle.version + 1,
        });

        compiled.push(primaryArticle.id);
      }

      // Also generate summaries for standalone articles
      for (const articleId of articleIds) {
        if (compiled.includes(articleId)) continue;

        const article = await articleRepo.findById(articleId);
        if (!article || article.summary) continue;

        const result = await compileArticle(
          [
            {
              id: article.id,
              title: article.title,
              contentMd: article.contentMd,
              category: article.category,
              equipmentBrand: article.equipmentBrand,
              equipmentModel: article.equipmentModel,
              procedureType: article.procedureType,
              sourceSessionIds: article.sourceSessionIds,
              version: article.version,
            },
          ],
          apiKey,
        );

        await articleRepo.update(article.id, { summary: result.summary });
        compiled.push(article.id);
      }

      return compiled;
    });

    // Step 3: Generate backlinks for all affected articles
    await step.run("generate-links", async () => {
      const articleRepo = new KnowledgeArticleRepository(db);
      const linkRepo = new KnowledgeLinkRepository(db);
      const deps = { articleRepo, linkRepo };

      for (const articleId of [...articleIds, ...compiledIds]) {
        await generateLinks(deps, articleId, orgId);
      }
    });

    // Step 4: Re-embed updated articles in Qdrant
    await step.run("re-embed", async () => {
      const apiKey = process.env.GEMINI_API_KEY!;
      const qdrantUrl = process.env.QDRANT_URL;
      if (!qdrantUrl) return;

      const articleRepo = new KnowledgeArticleRepository(db);
      const embeddingService = new GeminiEmbeddingService({ apiKey });
      const qdrantClient = createQdrantClient({
        url: qdrantUrl,
        apiKey: process.env.QDRANT_API_KEY,
      });

      for (const articleId of compiledIds) {
        const article = await articleRepo.findById(articleId);
        if (!article) continue;

        const embeddingText = `${article.title} ${article.summary ?? ""} ${article.contentMd}`;
        const denseVector = await embeddingService.embedDocument(embeddingText);

        await qdrantClient.upsert(QDRANT_CONFIG.COLLECTION_NAME, {
          wait: true,
          points: [
            {
              id: article.id,
              vectors: {
                [QDRANT_CONFIG.DENSE_VECTOR_NAME]: denseVector,
                [QDRANT_CONFIG.SPARSE_VECTOR_NAME]: {
                  text: embeddingText,
                  model: QDRANT_CONFIG.BM25_MODEL,
                },
              },
              payload: {
                article_id: article.id,
                org_id: article.orgId,
                confidence_level: article.confidenceLevel,
                equipment_brand: article.equipmentBrand ?? "",
                equipment_model: article.equipmentModel ?? "",
                category: article.category,
                title: article.title,
                summary: article.summary ?? "",
                tags: article.tags ?? [],
              },
            },
          ],
        });

        await articleRepo.updateEmbedding(article.id, article.id, EMBEDDING_CONFIG.MODEL);
      }
    });

    return { status: "completed", compiled: compiledIds.length };
  },
);
```
