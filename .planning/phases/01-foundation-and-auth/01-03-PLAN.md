       ```typescript
       import { initTRPC, TRPCError } from "@trpc/server";
       import { auth } from "@aucta/auth";

       export interface TRPCContext {
         session: {
           userId: string;
           activeOrganizationId: string | null;
         } | null;
         headers: Headers;
         ip: string | null;
       }

       export const createTRPCContext = async (opts: {
         headers: Headers;
       }): Promise<TRPCContext> => {
         const session = await auth.api.getSession({
           headers: opts.headers,
         });

         // Extract IP for rate limiting unauthenticated endpoints
         const ip = opts.headers.get("x-forwarded-for")?.split(",")[0]?.trim() ?? null;

         return {
           session: session
             ? {
                 userId: session.user.id,
                 activeOrganizationId: session.session.activeOrganizationId ?? null,
               }
             : null,
           headers: opts.headers,
           ip,
         };
       };

       const t = initTRPC.context<TRPCContext>().create();

       export const router = t.router;
       export const publicProcedure = t.procedure;
       export const middleware = t.middleware;
       ```
       Key: Use `initTRPC.context<TRPCContext>().create()` (tRPC v11 pattern). Do NOT use `createTRPCReact` (v10 pattern, deprecated).

       ```typescript
       import { TRPCError } from "@trpc/server";
       import { Ratelimit } from "@upstash/ratelimit";
       import { Redis } from "@upstash/redis";
       import { middleware } from "../trpc.js";

       const redis = Redis.fromEnv();

       export const authLimiter = new Ratelimit({
         redis,
         limiter: Ratelimit.fixedWindow(10, "60 s"),
         prefix: "rl:auth",
       });

       export const tokenLimiter = new Ratelimit({
         redis,
         limiter: Ratelimit.fixedWindow(5, "60 s"),
         prefix: "rl:token",
       });

       export const generalLimiter = new Ratelimit({
         redis,
         limiter: Ratelimit.tokenBucket(100, "60 s", 100),
         prefix: "rl:general",
       });

       export function createRateLimitMiddleware(limiter: Ratelimit) {
         return middleware(async (opts) => {
           const identifier =
             opts.ctx.session?.userId ?? opts.ctx.ip ?? "anonymous";
           const { success } = await limiter.limit(identifier);

           if (!success) {
             throw new TRPCError({
               code: "TOO_MANY_REQUESTS",
               message: "Rate limit exceeded. Try again later.",
             });
           }

           return opts.next();
         });
       }
       ```
       Note: Unauthenticated endpoints (login, signup) use IP as rate limit identifier (research pitfall 8).

       ```typescript
       import { TRPCError } from "@trpc/server";
       import { ErrorWithCode } from "@aucta/shared/utils/errors";
       import { middleware } from "../trpc.js";

       const ERROR_CODE_MAP: Record<string, TRPCError["code"]> = {
         NOT_FOUND: "NOT_FOUND",
         FORBIDDEN: "FORBIDDEN",
         BAD_REQUEST: "BAD_REQUEST",
         CONFLICT: "CONFLICT",
         UNAUTHORIZED: "UNAUTHORIZED",
       };

       export const errorConverter = middleware(async (opts) => {
         try {
           return await opts.next();
         } catch (error) {
           if (error instanceof ErrorWithCode) {
             throw new TRPCError({
               code: ERROR_CODE_MAP[error.code] ?? "INTERNAL_SERVER_ERROR",
               message: error.message,
             });
           }
           throw error;
         }
       });
       ```

    2. Create `packages/api/src/routers/invite.router.ts`:
       Uses orgProcedure for create/list/revoke (requires active org).
       Uses publicProcedure for lookup (unauthenticated users can look up codes).
       Uses protectedProcedure for join (authenticated but may not have org).

       Procedures:
       - `create` (orgProcedure + errorConverter):
         Input: CreateInviteInput from @aucta/shared (role, maxUses?, expiresInDays?)
         Permission: Check userRole is "owner" or ("manager" and role === "technician"). Managers per D-14 can only invite technicians.
         Action:
           a. Generate code via generateInviteCode()
           b. Generate id via generateId("inv")
           c. Calculate expiresAt from now + expiresInDays days
           d. Call inviteRepo.create({ id, orgId: ctx.orgId, code, createdBy: ctx.userId, role, maxUses, expiresAt })
           e. Call auditRepo.log(...)
           f. Return the created invite code DTO

       - `lookup` (publicProcedure + generalLimiter):
         Input: { code: z.string().length(6) }
         Action:
           a. Normalize code via normalizeInviteCode(input.code)
           b. Call inviteRepo.findByCode(normalized)
           c. If not found, throw ErrorWithCode.NotFound("This invite code is not valid...")
           d. Check expiry: if expiresAt < now, throw ErrorWithCode.BadRequest("This invite code has expired...")
           e. Check usage: if usedCount >= maxUses, throw ErrorWithCode.BadRequest("This invite code has reached its usage limit...")
           f. Get org details via `auth.api.getFullOrganization({ query: { organizationId: invite.orgId } })`
           g. Return LookupInviteResult: { code, orgName, role, memberCount }

       - `join` (protectedProcedure + errorConverter):
         Input: { code: z.string().length(6) }

           a. Normalize and look up code (same validation as lookup)

           c. If user has org, throw ErrorWithCode.Conflict("You already belong to an organization")
           d. Call `auth.api.addMember({ body: { organizationId: invite.orgId, userId: ctx.session.userId, role: invite.role } })`
           e. Call `auth.api.setActiveOrganization({ body: { organizationId: invite.orgId } })`
           f. Call inviteRepo.incrementUsage(invite.id)
           g. Call auditRepo.log(...)
           h. Return success with org details

         No input.
         Action: Call inviteRepo.findByOrg(ctx.orgId)

         Input: { inviteId: z.string() }
         Permission: userRole must be "owner"

           a. Call inviteRepo.delete(input.inviteId)
           b. Call auditRepo.log(...)
