    3. Create `packages/auth/src/permissions.ts` per D-14:
       ```typescript
       import { createAccessControl } from "better-auth/plugins/access";

       const statement = {
         organization: ["update", "delete"],
         member: ["create", "update", "delete"],
         invitation: ["create", "cancel"],
         invite: ["create", "revoke", "view"],
         settings: ["view", "update"],
         team: ["view"],
       } as const;

       export const ac = createAccessControl(statement);

       // D-14: Technician — can use the app and view own profile only
       export const technician = ac.newRole({
         team: ["view"],
       });

       // D-14: Manager — invite technicians, view team, view own profile
       export const manager = ac.newRole({
         invite: ["create", "view"],
         team: ["view"],
         member: ["create"],
       });

       // D-14: Owner — full control over org, invites, members, settings
       export const owner = ac.newRole({
         organization: ["update", "delete"],
         member: ["create", "update", "delete"],
         invitation: ["create", "cancel"],
         invite: ["create", "revoke", "view"],
         settings: ["view", "update"],
         team: ["view"],
       });
       ```

    4. Create `packages/auth/src/server.ts`:
       ```typescript
       import { betterAuth } from "better-auth";
       import { drizzleAdapter } from "better-auth/adapters/drizzle";
       import { organization } from "better-auth/plugins";
       import { db } from "@aucta/db/client";
       import { ac, owner, manager, technician } from "./permissions.js";

       export const auth = betterAuth({
         database: drizzleAdapter(db, { provider: "pg" }),
         emailAndPassword: { enabled: true },
         trustedOrigins: ["aucta://"],
         plugins: [
           organization({
             ac,
             roles: { owner, manager, technician },
             membershipLimit: 100,
             organizationLimit: 1,   // D-12: single org per user
             creatorRole: "owner",   // D-13: org creator becomes owner
           }),
         ],
         session: {
           expiresIn: 60 * 60 * 24 * 30,  // 30 days
           updateAge: 60 * 60 * 24,        // refresh daily
         },
       });

       export type Auth = typeof auth;
       ```
       Note: Better Auth automatically creates its own tables (user, session, account, verification, organization, member, invitation) via the Drizzle adapter on first use.
