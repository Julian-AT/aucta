export class MockWebSocket {
  static CONNECTING = 0;
  static OPEN = 1;
  static CLOSING = 2;
  static CLOSED = 3;

  readyState = MockWebSocket.CONNECTING;
  url: string;
  sentMessages: string[] = [];

  onopen: ((event: any) => void) | null = null;
  onmessage: ((event: any) => void) | null = null;
  onclose: ((event: any) => void) | null = null;
  onerror: ((event: any) => void) | null = null;

  constructor(url: string) {
    this.url = url;
  }

  send(data: string): void {
    if (this.readyState !== MockWebSocket.OPEN) {
      throw new Error("WebSocket is not open");
    }
    this.sentMessages.push(data);
  }

  close(code?: number, reason?: string): void {
    this.readyState = MockWebSocket.CLOSED;
    this.onclose?.({ code: code ?? 1000, reason: reason ?? "" });
  }

  // Test helpers
  simulateOpen(): void {
    this.readyState = MockWebSocket.OPEN;
    this.onopen?.({});
  }

  simulateMessage(data: unknown): void {
    this.onmessage?.({ data: JSON.stringify(data) });
  }

  simulateError(error: string): void {
    this.onerror?.({ message: error });
  }

  simulateClose(code = 1000, reason = ""): void {
    this.readyState = MockWebSocket.CLOSED;
    this.onclose?.({ code, reason });
  }
}

**Create apps/mobile/stores/session-store.ts:**

The Zustand store manages the session lifecycle as a state machine. WebSocket callbacks and native audio callbacks fire outside React, so getState()/setState() must work without React context. This is why Zustand was chosen over useState/useReducer.

State fields:
- `status`: SessionStatus - one of "idle", "requesting-permissions", "disclosure", "connecting", "active", "disconnected", "reconnecting", "completed", "failed", "timeout" (per UI-SPEC State-Driven UI Matrix)
- `connectionQuality`: "connected" | "reconnecting" | "lost" (per D-15 traffic light)
- `sessionId`: string | null
- `resumptionHandle`: string | null (Gemini sessionResumption handle, updated via SessionResumptionUpdate messages)
- `isMicActive`: boolean (always-on mic status per D-05)
- `isAISpeaking`: boolean (drives AudioIndicator animation per D-04)
- `reconnectAttempts`: number (tracked for backoff calculation)
- `error`: string | null
- `sessionStartTime`: number | null (Date.now() when session becomes active, for token expiry tracking)

Actions: setStatus, setConnectionQuality, setSessionId, setResumptionHandle, setMicActive, setAISpeaking, incrementReconnectAttempts, resetReconnectAttempts, setError, startSession (sets status + sessionStartTime), reset (returns to initial state).
