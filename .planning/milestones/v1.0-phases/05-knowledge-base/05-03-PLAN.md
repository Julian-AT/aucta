Define the hybrid search function with constructor-injected dependencies:

```typescript
import type { QdrantClient } from "@qdrant/js-client-rest";
import type { GeminiEmbeddingService } from "./embeddings.js";
import type { KnowledgeSearchResult, KnowledgeSearchOptions } from "../core/types.js";
import { QDRANT_CONFIG } from "../core/constants.js";
import { getAttributionPhrase } from "./context-builder.js";

export interface HybridSearchDeps {
  qdrantClient: QdrantClient;
  embeddingService: GeminiEmbeddingService;
}

export async function hybridSearch(
  deps: HybridSearchDeps,
  options: KnowledgeSearchOptions,
): Promise<KnowledgeSearchResult[]> {
  const { qdrantClient, embeddingService } = deps;
  const limit = options.limit ?? QDRANT_CONFIG.DEFAULT_SEARCH_LIMIT;

  // Generate dense embedding for the query
  const denseVector = await embeddingService.embedQuery(options.query);

  // Build filter -- org_id is ALWAYS required (tenant isolation)
  const mustFilters: Array<Record<string, unknown>> = [
    { key: "org_id", match: { value: options.orgId } },
  ];
  if (options.equipmentBrand) {
    mustFilters.push({ key: "equipment_brand", match: { value: options.equipmentBrand } });
  }
  if (options.equipmentModel) {
    mustFilters.push({ key: "equipment_model", match: { value: options.equipmentModel } });
  }

  // Hybrid query: prefetch dense + BM25, fuse with weighted RRF
  // Weight dense higher (2.0) than sparse (1.0) for semantic relevance (per research)
  const results = await qdrantClient.query(QDRANT_CONFIG.COLLECTION_NAME, {
    prefetch: [
      {
        query: { text: options.query, model: QDRANT_CONFIG.BM25_MODEL },
        using: QDRANT_CONFIG.SPARSE_VECTOR_NAME,
        limit: 20,
      },
      {
        query: denseVector,
        using: QDRANT_CONFIG.DENSE_VECTOR_NAME,
        limit: 20,
      },
    ],
    query: { rrf: { weights: [1.0, 2.0] } },
    filter: { must: mustFilters },
    with_payload: true,
    limit,
  });

  return (results.points ?? []).map((point) => {
    const payload = point.payload as Record<string, unknown>;
    const confidenceLevel = (payload.confidence_level as string) ?? "unverified";
    return {
      id: String(point.id),
      title: String(payload.title ?? ""),
      summary: payload.summary ? String(payload.summary) : null,
      contentMd: String(payload.content_md ?? ""),
      confidenceLevel: confidenceLevel as KnowledgeSearchResult["confidenceLevel"],
      equipmentBrand: payload.equipment_brand ? String(payload.equipment_brand) : null,
      equipmentModel: payload.equipment_model ? String(payload.equipment_model) : null,
      procedureType: payload.procedure_type ? String(payload.procedure_type) : null,
      category: (payload.category as string) ?? "equipment",
      score: point.score ?? 0,
      attribution: getAttributionPhrase(confidenceLevel),
    };
  });
}
```

```typescript
import { CONFIDENCE_ATTRIBUTION } from "../core/constants.js";
import type { KnowledgeSearchResult } from "../core/types.js";
import type { PromptLayer } from "../../prompts/builder.js";

export function getAttributionPhrase(confidenceLevel: string): string {
  return (
    CONFIDENCE_ATTRIBUTION[confidenceLevel as keyof typeof CONFIDENCE_ATTRIBUTION] ??
    CONFIDENCE_ATTRIBUTION.unverified
  );
}

export function buildKnowledgeContext(results: KnowledgeSearchResult[]): PromptLayer {
  if (results.length === 0) {
    return {
      id: "knowledge-context",
      content: "",
      required: false,
    };
  }

  const formattedResults = results
    .map((r, i) => {
      const equipment = [r.equipmentBrand, r.equipmentModel].filter(Boolean).join(" ");
      return `[${i + 1}] ${r.title}${equipment ? ` (${equipment})` : ""}
Confidence: ${r.attribution}
${r.contentMd}`;
    })
    .join("\n\n---\n\n");

  const content = `KNOWLEDGE BASE RESULTS:
When referencing knowledge base information, naturally attribute the source confidence level to the technician.
Use phrases like "${CONFIDENCE_ATTRIBUTION.expert_approved}" for verified content or "${CONFIDENCE_ATTRIBUTION.unverified}" for unverified observations.

${formattedResults}`;

  return {
    id: "knowledge-context",
    content,
    required: false,
  };
}
```

```typescript
import { buildTool } from "../base.js";
import { hybridSearch, type HybridSearchDeps } from "../../knowledge/retrieve/search.js";
import { getAttributionPhrase } from "../../knowledge/retrieve/context-builder.js";
import {
  SearchKBInput,
  SearchKBOutput,
  type SearchKBInputType,
  type SearchKBOutputType,
} from "./types.js";

export function createSearchKnowledgeBaseTool(deps: HybridSearchDeps) {
  return buildTool<SearchKBInputType, SearchKBOutputType>({
    name: "searchKnowledgeBase",
    description:
      "Search the organization knowledge base for HVAC procedures, equipment information, troubleshooting guides, and verified knowledge articles",
    inputSchema: SearchKBInput,
    outputSchema: SearchKBOutput,
    isReadOnly: true,
    async call(input, context): Promise<SearchKBOutputType> {
      const results = await hybridSearch(deps, {
        query: input.query,
        orgId: context.orgId,
        equipmentBrand: input.equipmentBrand,
        equipmentModel: input.equipmentModel,
        limit: 3,
      });

      if (results.length === 0) {
        return {
          found: false,
          message: "No relevant knowledge articles found in the organization knowledge base. I'll provide guidance based on general HVAC expertise.",
        };
      }

      return {
        found: true,
        articles: results.map((r) => ({
          title: r.title,
          summary: r.summary,
          content: r.contentMd,
          confidenceLevel: r.confidenceLevel,
          equipmentBrand: r.equipmentBrand,
          equipmentModel: r.equipmentModel,
          attribution: r.attribution,
        })),
      };
    },
  });
}
```
