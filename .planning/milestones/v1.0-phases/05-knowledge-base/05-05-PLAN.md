```typescript
import type { HealthCheckResult } from "../core/types.js";
import type { KnowledgeArticleRepository } from "@aucta/db/repositories/knowledge-article.repository";
import type { DrizzleClient } from "@aucta/db/client";
import { knowledgeArticles } from "@aucta/db/schema";
import { aiSessions } from "@aucta/db/schema";
import { sql, eq, and, isNotNull, notInArray, count } from "drizzle-orm";

const DEFAULT_STALE_DAYS = 90;

export interface HealthCheckDeps {
  articleRepo: KnowledgeArticleRepository;
  db: DrizzleClient;
}

/**
 * Run health checks for an org's knowledge base.
 * Detects: coverage gaps, stale articles, conflicting articles.
 */
export async function runHealthCheck(
  deps: HealthCheckDeps,
  orgId: string,
  staleDays: number = DEFAULT_STALE_DAYS,
): Promise<HealthCheckResult> {
  const [coverageGaps, staleArticles, conflictingArticles] = await Promise.all([
    detectCoverageGaps(deps.db, orgId),
    detectStaleArticles(deps.articleRepo, orgId, staleDays),
    detectConflicts(deps.db, orgId),
  ]);

  return {
    coverageGaps,
    staleArticles,
    conflictingArticles,
    checkedAt: new Date(),
  };
}

/**
 * Find equipment models that appear in sessions but have no knowledge articles.
 * Uses SQL: SELECT DISTINCT manufacturer, equipmentModel FROM ai_session_findings
 * WHERE orgId = ? AND NOT EXISTS (SELECT 1 FROM knowledge_articles WHERE ...)
 */
async function detectCoverageGaps(
  db: DrizzleClient,
  orgId: string,
): Promise<HealthCheckResult["coverageGaps"]> {
  // Query sessions that have equipment info but no matching knowledge article
  // Uses raw SQL for the NOT EXISTS subquery which is cleaner for this pattern
  const result = await db.execute(sql`
    SELECT
      f.manufacturer AS equipment_brand,
      f.equipment_model AS equipment_model,
      COUNT(DISTINCT s.id) AS session_count
    FROM ai_session_findings f
    JOIN ai_sessions s ON s.id = f.session_id
    WHERE s.org_id = ${orgId}
      AND f.equipment_model IS NOT NULL
      AND NOT EXISTS (
        SELECT 1 FROM knowledge_articles ka
        WHERE ka.org_id = ${orgId}
          AND ka.equipment_model = f.equipment_model
          AND (ka.equipment_brand = f.manufacturer OR f.manufacturer IS NULL)
      )
    GROUP BY f.manufacturer, f.equipment_model
    ORDER BY session_count DESC
  `);

  return (result.rows ?? []).map((row: Record<string, unknown>) => ({
    equipmentBrand: String(row.equipment_brand ?? ""),
    equipmentModel: String(row.equipment_model ?? ""),
    sessionCount: Number(row.session_count ?? 0),
  }));
}

/**
 * Find articles not updated in staleDays+ days.
 */
async function detectStaleArticles(
  articleRepo: KnowledgeArticleRepository,
  orgId: string,
  staleDays: number,
): Promise<HealthCheckResult["staleArticles"]> {
  const stale = await articleRepo.findStaleArticles(orgId, staleDays);

  return stale.map((article) => ({
    articleId: article.id,
    title: article.title,
    daysSinceUpdate: Math.floor(
      (Date.now() - new Date(article.updatedAt).getTime()) / (1000 * 60 * 60 * 24),
    ),
  }));
}

/**
 * Find articles that cover the same equipment + procedure type (potential conflicts).
 * Groups by (equipmentBrand, equipmentModel, procedureType) where count > 1.
 */
async function detectConflicts(
  db: DrizzleClient,
  orgId: string,
): Promise<HealthCheckResult["conflictingArticles"]> {
  const result = await db.execute(sql`
    SELECT
      CONCAT(equipment_brand, '/', equipment_model, '/', procedure_type) AS procedure_key,
      ARRAY_AGG(id) AS article_ids
    FROM knowledge_articles
    WHERE org_id = ${orgId}
      AND procedure_type IS NOT NULL
      AND equipment_brand IS NOT NULL
    GROUP BY equipment_brand, equipment_model, procedure_type
    HAVING COUNT(*) > 1
    ORDER BY COUNT(*) DESC
  `);

  return (result.rows ?? []).map((row: Record<string, unknown>) => ({
    procedureKey: String(row.procedure_key ?? ""),
    articleIds: Array.isArray(row.article_ids) ? row.article_ids.map(String) : [],
  }));
}
```

```typescript
import { runHealthCheck } from "@aucta/ai/knowledge/health/health-check";
import { KnowledgeArticleRepository } from "@aucta/db/repositories/knowledge-article.repository";
import { db } from "@aucta/db/client";
import { inngest } from "./inngest-client.js";

export const knowledgeHealth = inngest.createFunction(
  {
    id: "knowledge-health",
    retries: 2,
    triggers: [{ cron: "0 3 * * *" }], // Daily at 3 AM UTC
  },
  async ({ step }) => {
    // Step 1: Get all orgs with knowledge articles
    const orgIds = await step.run("get-active-orgs", async () => {
      const result = await db.execute(
        sql`SELECT DISTINCT org_id FROM knowledge_articles`
      );
      return (result.rows ?? []).map((row: Record<string, unknown>) => String(row.org_id));
    });

    if (orgIds.length === 0) return { status: "no-orgs" };

    // Step 2: Run health check for each org
    const results: Array<{ orgId: string; gaps: number; stale: number; conflicts: number }> = [];

    for (const orgId of orgIds) {
      const result = await step.run(`health-check-${orgId}`, async () => {
        const articleRepo = new KnowledgeArticleRepository(db);
        return runHealthCheck({ articleRepo, db }, orgId);
      });

      results.push({
        orgId,
        gaps: result.coverageGaps.length,
        stale: result.staleArticles.length,
        conflicts: result.conflictingArticles.length,
      });
    }

    // Step 3: Send event with aggregated results (for future dashboard/alerts)
    if (results.some((r) => r.gaps > 0 || r.stale > 0 || r.conflicts > 0)) {
      await step.run("emit-results", async () => {
        await inngest.send({
          name: "knowledge/health-check-complete",
          data: { results, checkedAt: new Date().toISOString() },
        });
      });
    }

    return { status: "completed", orgCount: orgIds.length, results };
  },
);
```

```typescript
import { z } from "zod";
import { TRPCError } from "@trpc/server";
import { router, orgProcedure } from "../trpc.js";
import { KnowledgeArticleRepository } from "@aucta/db/repositories/knowledge-article.repository";
import { KnowledgeLinkRepository } from "@aucta/db/repositories/knowledge-link.repository";
import { GeminiEmbeddingService, createQdrantClient, QDRANT_CONFIG, EMBEDDING_CONFIG } from "@aucta/ai/knowledge";
import { generateLinks } from "@aucta/ai/knowledge/compile/linker";
import { runHealthCheck } from "@aucta/ai/knowledge/health/health-check";

export const knowledgeRouter = router({
  create: orgProcedure
    .input(z.object({
      title: z.string().min(1).max(500),
      contentMd: z.string().min(1),
      category: z.enum(["equipment", "procedure", "safety", "regulatory", "troubleshooting"]),
      equipmentBrand: z.string().optional(),
      equipmentModel: z.string().optional(),
      procedureType: z.string().optional(),
      tags: z.array(z.string()).optional(),
    }))
    .mutation(async ({ ctx, input }) => {
      const articleRepo = new KnowledgeArticleRepository(ctx.db);
      const orgId = ctx.session.user.orgId;

      // Create article with manual source and technician_reviewed confidence
      const article = await articleRepo.create({
        orgId,
        title: input.title,
        contentMd: input.contentMd,
        category: input.category,
        equipmentBrand: input.equipmentBrand,
        equipmentModel: input.equipmentModel,
        procedureType: input.procedureType,
        sourceType: "manual",
        confidenceLevel: "technician_reviewed",
        tags: input.tags,
        isAiGenerated: false,
        createdBy: ctx.session.user.id,
      });

      // Auto-embed and upsert to Qdrant
      await embedAndUpsert(article, ctx.db);

      // Auto-generate backlinks
      const linkRepo = new KnowledgeLinkRepository(ctx.db);
      await generateLinks({ articleRepo, linkRepo }, article.id, orgId);

      return article;
    }),

  update: orgProcedure
    .input(z.object({
      id: z.string().uuid(),
      title: z.string().min(1).max(500).optional(),
      contentMd: z.string().min(1).optional(),
      category: z.enum(["equipment", "procedure", "safety", "regulatory", "troubleshooting"]).optional(),
      equipmentBrand: z.string().optional(),
      equipmentModel: z.string().optional(),
      procedureType: z.string().optional(),
      tags: z.array(z.string()).optional(),
    }))
    .mutation(async ({ ctx, input }) => {
      const articleRepo = new KnowledgeArticleRepository(ctx.db);
      const orgId = ctx.session.user.orgId;

      const existing = await articleRepo.findById(input.id);
      if (!existing) {
        throw new TRPCError({ code: "NOT_FOUND", message: `Article ${input.id} not found` });
      }
      if (existing.orgId !== orgId) {
        throw new TRPCError({ code: "FORBIDDEN", message: "Cross-org access denied" });
      }

      const { id, ...updateData } = input;
      const updated = await articleRepo.update(id, updateData);

      // Re-embed and re-upsert to Qdrant
      await embedAndUpsert(updated, ctx.db);

      // Re-generate backlinks
      const linkRepo = new KnowledgeLinkRepository(ctx.db);
      await generateLinks({ articleRepo, linkRepo }, updated.id, orgId);

      return updated;
    }),

  list: orgProcedure
    .input(z.object({
      limit: z.number().min(1).max(100).default(20),
      offset: z.number().min(0).default(0),
      equipmentBrand: z.string().optional(),
      category: z.enum(["equipment", "procedure", "safety", "regulatory", "troubleshooting"]).optional(),
      confidenceLevel: z.enum(["unverified", "technician_reviewed", "expert_approved", "manufacturer_sourced"]).optional(),
    }).optional())
    .query(async ({ ctx, input }) => {
      const articleRepo = new KnowledgeArticleRepository(ctx.db);
      const orgId = ctx.session.user.orgId;
      const opts = input ?? {};

      // Use findByOrgId with filters
      const articles = await articleRepo.findByOrgId(orgId, {
        limit: opts.limit ?? 20,
        offset: opts.offset ?? 0,
      });

      // Client-side filter for now (repository can be enhanced later)
      return articles.filter((a) => {
        if (opts.equipmentBrand && a.equipmentBrand !== opts.equipmentBrand) return false;
        if (opts.category && a.category !== opts.category) return false;
        if (opts.confidenceLevel && a.confidenceLevel !== opts.confidenceLevel) return false;
        return true;
      });
    }),

  getById: orgProcedure
    .input(z.object({ id: z.string().uuid() }))
    .query(async ({ ctx, input }) => {
      const articleRepo = new KnowledgeArticleRepository(ctx.db);
      const orgId = ctx.session.user.orgId;

      const article = await articleRepo.findById(input.id);
      if (!article) {
        throw new TRPCError({ code: "NOT_FOUND", message: `Article ${input.id} not found` });
      }
      if (article.orgId !== orgId) {
        throw new TRPCError({ code: "FORBIDDEN", message: "Cross-org access denied" });
      }

      return article;
    }),

  healthCheck: orgProcedure
    .query(async ({ ctx }) => {
      const articleRepo = new KnowledgeArticleRepository(ctx.db);
      const orgId = ctx.session.user.orgId;

      return runHealthCheck({ articleRepo, db: ctx.db }, orgId);

/**
 * Embed article text and upsert to Qdrant.
 * Silent failure — article is in Postgres regardless.
 */
async function embedAndUpsert(
  article: { id: string; orgId: string; title: string; summary?: string | null; contentMd: string; confidenceLevel: string; equipmentBrand?: string | null; equipmentModel?: string | null; category: string; tags?: string[] | null },
  db: any,
): Promise<void> {
  const apiKey = process.env.GEMINI_API_KEY;
  const qdrantUrl = process.env.QDRANT_URL;
  if (!apiKey || !qdrantUrl) return;

  try {
    const embeddingService = new GeminiEmbeddingService({ apiKey });
    const qdrantClient = createQdrantClient({
      url: qdrantUrl,
      apiKey: process.env.QDRANT_API_KEY,
    });

    const embeddingText = `${article.title} ${article.summary ?? ""} ${article.contentMd}`;
    const denseVector = await embeddingService.embedDocument(embeddingText);

    await qdrantClient.upsert(QDRANT_CONFIG.COLLECTION_NAME, {
      wait: true,
      points: [{
        id: article.id,
        vectors: {
          [QDRANT_CONFIG.DENSE_VECTOR_NAME]: denseVector,
          [QDRANT_CONFIG.SPARSE_VECTOR_NAME]: {
            text: embeddingText,
            model: QDRANT_CONFIG.BM25_MODEL,
          },
        },
        payload: {
          article_id: article.id,
          org_id: article.orgId,
          confidence_level: article.confidenceLevel,
          equipment_brand: article.equipmentBrand ?? "",
          equipment_model: article.equipmentModel ?? "",
          category: article.category,
          title: article.title,
          summary: article.summary ?? "",
          tags: article.tags ?? [],
        },
      }],
    });

    const articleRepo = new KnowledgeArticleRepository(db);
    await articleRepo.updateEmbedding(article.id, article.id, EMBEDDING_CONFIG.MODEL);
  } catch (err) {
    console.error(`Failed to embed article ${article.id}:`, err);
  }
}
```
