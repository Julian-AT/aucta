```typescript
import { QdrantClient } from "@qdrant/js-client-rest";
import { QDRANT_CONFIG, EMBEDDING_CONFIG } from "../core/constants.js";

export interface QdrantClientConfig {
  url: string;
  apiKey?: string;
}

export function createQdrantClient(config: QdrantClientConfig): QdrantClient {
  return new QdrantClient({
    url: config.url,
    apiKey: config.apiKey,
  });
}

export async function ensureCollection(client: QdrantClient): Promise<void> {
  const collections = await client.getCollections();
  const exists = collections.collections.some(
    (c) => c.name === QDRANT_CONFIG.COLLECTION_NAME,
  );

  if (exists) return;

  await client.createCollection(QDRANT_CONFIG.COLLECTION_NAME, {
    vectors: {
      [QDRANT_CONFIG.DENSE_VECTOR_NAME]: {
        size: EMBEDDING_CONFIG.DIMENSIONS,
        distance: "Cosine",
      },
    },
    sparse_vectors: {
      [QDRANT_CONFIG.SPARSE_VECTOR_NAME]: {
        modifier: "idf",
      },
    },
  });

  // Create payload indexes for efficient filtering (per D-12)
  const indexFields = [
    { field_name: "org_id", field_schema: "keyword" as const },
    { field_name: "confidence_level", field_schema: "keyword" as const },
    { field_name: "equipment_brand", field_schema: "keyword" as const },
    { field_name: "equipment_model", field_schema: "keyword" as const },
    { field_name: "category", field_schema: "keyword" as const },
  ];

  for (const idx of indexFields) {
    await client.createPayloadIndex(QDRANT_CONFIG.COLLECTION_NAME, idx);
  }
}
```

**File 2: packages/ai/src/knowledge/retrieve/embeddings.ts**

```typescript
import { GoogleGenAI } from "@google/genai";
import { EMBEDDING_CONFIG } from "../core/constants.js";

export interface EmbeddingServiceConfig {
  apiKey: string;
}

export class GeminiEmbeddingService {
  private ai: GoogleGenAI;

  constructor(private config: EmbeddingServiceConfig) {
    this.ai = new GoogleGenAI({ apiKey: config.apiKey });
  }

  async embedDocument(text: string): Promise<number[]> {
    const truncated = this.truncateIfNeeded(text);
    const result = await this.ai.models.embedContent({
      model: EMBEDDING_CONFIG.MODEL,
      contents: [truncated],
      config: {
        outputDimensionality: EMBEDDING_CONFIG.DIMENSIONS,
        taskType: "RETRIEVAL_DOCUMENT",
      },
    });
    return result.embeddings![0]!.values!;
  }

  async embedQuery(text: string): Promise<number[]> {
    const result = await this.ai.models.embedContent({
      model: EMBEDDING_CONFIG.MODEL,
      contents: [text],
      config: {
        outputDimensionality: EMBEDDING_CONFIG.DIMENSIONS,
        taskType: "RETRIEVAL_QUERY",
      },
    });
    return result.embeddings![0]!.values!;
  }

  /**
   * Truncate text if it likely exceeds the model's token limit.
   * Rough estimate: 1 token per 4 chars for English text.
   * For oversized articles, use title + summary as compact representation.
   */
  private truncateIfNeeded(text: string): string {
    const estimatedTokens = Math.ceil(text.length / EMBEDDING_CONFIG.CHARS_PER_TOKEN_ESTIMATE);
    if (estimatedTokens <= EMBEDDING_CONFIG.MAX_TOKENS) return text;

    // Truncate to approximately MAX_TOKENS worth of characters
    const maxChars = EMBEDDING_CONFIG.MAX_TOKENS * EMBEDDING_CONFIG.CHARS_PER_TOKEN_ESTIMATE;
    return text.slice(0, maxChars);
  }
}
```
