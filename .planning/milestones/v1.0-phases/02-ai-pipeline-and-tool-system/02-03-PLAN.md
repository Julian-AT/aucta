       ```typescript
       import type { ToolDef, GeminiFunctionDeclaration } from "./base.js";

       export interface ToolPool {
         tools: Map<string, ToolDef<unknown, unknown>>;
         functionDeclarations: GeminiFunctionDeclaration[];
         getTool(name: string): ToolDef<unknown, unknown> | undefined;
       }

       export function assembleToolPool(
         tools: ToolDef<unknown, unknown>[]
       ): ToolPool {
         const toolMap = new Map<string, ToolDef<unknown, unknown>>();

         for (const tool of tools) {
           if (toolMap.has(tool.name)) {
             throw new Error(`Duplicate tool name: ${tool.name}`);
           }
           toolMap.set(tool.name, tool);
         }

         return {
           tools: toolMap,
           functionDeclarations: tools.map((t) => t.prompt()),
           getTool: (name: string) => toolMap.get(name),
         };
       }
       ```

    2. Create packages/ai/src/tools/dispatcher.ts:
       ```typescript
       import type { ToolPool } from "./registry.js";
       import type { ToolContext, ToolResult } from "../types.js";
       import { ZodError } from "zod";

       export interface ToolCallEvent {
         functionCalls: Array<{
           id: string;
           name: string;
           args: Record<string, unknown>;
         }>;
       }

       export interface ToolResponseMessage {
         functionResponses: Array<{
           id: string;
           name: string;
           response: Record<string, unknown>;
         }>;
       }

       class TimeoutError extends Error {
         constructor() { super("Tool execution timed out"); this.name = "TimeoutError"; }
       }

       const DEFAULT_TIMEOUT_MS = 10_000; // D-11

       function timeoutPromise(ms: number): Promise<never> {
         return new Promise((_, reject) => {
           setTimeout(() => reject(new TimeoutError()), ms);
         });
       }

       export async function dispatchToolCalls(
         toolCall: ToolCallEvent,
         pool: ToolPool,
         context: ToolContext,
         timeoutMs = DEFAULT_TIMEOUT_MS,
       ): Promise<ToolResponseMessage> {
         const functionResponses = await Promise.all(
           toolCall.functionCalls.map(async (fc) => {
             const tool = pool.getTool(fc.name);

             if (!tool) {
               return {
                 id: fc.id,
                 name: fc.name,
                 response: { error: `Unknown tool: ${fc.name}`, code: "TOOL_NOT_FOUND" },
               };
             }

             try {
               const validatedInput = tool.validateInput(fc.args);
               const result = await Promise.race([
                 tool.call(validatedInput, context),
                 timeoutPromise(timeoutMs),
               ]);

               if (result.success) {
                 return { id: fc.id, name: fc.name, response: result.data as Record<string, unknown> };
               }
               return { id: fc.id, name: fc.name, response: { error: result.error, code: result.code } };
             } catch (error) {
               if (error instanceof ZodError) {
                 return {
                   id: fc.id,
                   name: fc.name,
                   response: {
                     error: `Invalid input: ${error.issues.map((i) => i.message).join(", ")}`,
                     code: "VALIDATION_ERROR",
                   },
                 };
               }
               if (error instanceof TimeoutError) {
                 return {
                   id: fc.id,
                   name: fc.name,
                   response: { error: "Tool execution timed out", code: "TIMEOUT" },
                 };
               }
               return {
                 id: fc.id,
                 name: fc.name,
                 response: { error: "Internal tool error", code: "INTERNAL_ERROR" },
                 };
             }
           }),
         );

         return { functionResponses };
       }
       ```

       ```typescript
       import { z } from "zod";
       import { router } from "../trpc.js";
       import { orgProcedure } from "../middleware/org-scope.js";
       import { createRateLimitMiddleware, tokenLimiter } from "../middleware/rate-limit.js";
       import { errorConverter } from "../middleware/error-converter.js";
       import { GoogleGenAI } from "@google/genai";
       import {
         buildEffectiveSystemPrompt,
         coreRolePrompt,
         hvacExpertisePrompt,
         safetyRulesPrompt,
         assembleToolPool,
       } from "@aucta/ai";
       // Import the default tool instances from packages/ai
       // (lookupErrorCode with JsonErrorCodeRepository, logFinding with InMemoryFindingStore)

       const GEMINI_LIVE_MODEL = process.env.GEMINI_LIVE_MODEL ?? "gemini-2.5-flash-live-preview";

       export const sessionRouter = router({
         createToken: orgProcedure
           .use(createRateLimitMiddleware(tokenLimiter))
           .use(errorConverter)
           .mutation(async ({ ctx }) => {
             const apiKey = process.env.GEMINI_API_KEY;
             if (!apiKey) {
               throw new Error("GEMINI_API_KEY environment variable is not set");
             }

             // 1. Build system prompt from layers (TOOL-08)
             const systemPrompt = buildEffectiveSystemPrompt([
               coreRolePrompt,
               hvacExpertisePrompt,
               safetyRulesPrompt,
             ]);

             // 2. Assemble tool pool for function declarations (TOOL-02)
             const pool = assembleToolPool([
               lookupErrorCodeTool,
               logFindingTool,
             ]);

             // 3. Create ephemeral token via Gemini SDK
             const client = new GoogleGenAI({ apiKey });
             const tokenResponse = await client.authTokens.create({
               config: {
                 uses: 1,
                 expireTime: new Date(Date.now() + 30 * 60 * 1000).toISOString(),
                 newSessionExpireTime: new Date(Date.now() + 60 * 1000).toISOString(),
                 httpOptions: { apiVersion: "v1alpha" },
                 liveConnectConstraints: {
                   model: GEMINI_LIVE_MODEL,
                   config: {
                     systemInstruction: systemPrompt,
                     responseModalities: ["AUDIO"],
                   },
                 },
               },
             });

             // NEVER log tokenResponse.name (the ephemeral API key)
             return {
               token: tokenResponse.name,
               model: GEMINI_LIVE_MODEL,
               toolDeclarations: pool.functionDeclarations,
             };
           }),
       });
       ```
